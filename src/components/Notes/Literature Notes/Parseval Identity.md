>$$
\begin{align}
\forall f &\in \mathcal{E}, f=\sum_{k=1}^{n} (e_{k},f)e_{k} \\
\forall f&,g \in \mathcal{E} , (f,g)=\sum_{i=1}^{n}(f,e_{k})(e_{k},g) \\
\forall f& \in \mathcal{E}, \lVert f \rVert ^{2}=\sum_{i=1}^{n} \lvert (e_{k},f)^{2} \rvert  
\end{align}$$
